// Closed-loop PID Drive Controller with Heading Correction

void drivePID(double targetDist, double targetHeading) {

double kP = 0.48, kD = 0.18, kP_turn = 0.85;

while(true) {

double currentPos = (LeftEncoder.get_value() + RightEncoder.get_value()) / 2.0;

double error = targetDist - currentPos;

double headingError = targetHeading - InertialSensor.get_heading();

double power = (error * kP) + ((error - prevError) * kD);

double turnCorrection = headingError * kP_turn;

LeftMotors.move_voltage(power + turnCorrection);

RightMotors.move_voltage(power - turnCorrection);

if(fabs(error) < 0.3) break;

pros::delay(10);

}

}
